 const goNext=()=>{
  if(step===0 && !user.firstName.trim()){alert("Please enter your first name.");return}

export function Diagnostic(){
 const navigate=useNavigate(); const [state,setState]=useState(readState()); const [section,setSection]=useState(firstIncompleteDiagnosticSection(state.diagnostic.completedSections)); const [qIndex,setQIndex]=useState(state.diagnostic.currentQuestion||0); const [showProcessing,setShowProcessing]=useState(false); const [writing,setWriting]=useState(state.diagnostic.writingText); const [speakingTime,setSpeakingTime]=useState(state.diagnostic.speakingDuration||0); const audioRef=useRef<HTMLAudioElement>(null);
 const current=diagnosticSections[section]; const completed=new Set(state.diagnostic.completedSections);
 const flatTotal=diagnosticSections.reduce((n,s)=>n+(s.questions?.length??1),0); const answeredObj=Object.keys(state.diagnostic.responses).length; const progress=Math.round(answeredObj/28*100);
 useEffect(()=>{updateState(s=>{s.currentActivity={mode:"TEST",title:"Diagnostic Assessment",route:"/diagnostic"}})},[]);
 useEffect(()=>{if(state.diagnostic.status==="not_started")updateState(s=>{s.diagnostic.status="in_progress";s.diagnostic.startedAt=new Date().toISOString()})},[]);
 const local=s=>setState(readState());

 const finishSection=()=>{
  persist(s=>{if(!s.diagnostic.completedSections.includes(current.id))s.diagnostic.completedSections.push(current.id);s.diagnostic.currentSection=Math.min(diagnosticSections.length-1,section+1);s.diagnostic.currentQuestion=0});
  if(section<diagnosticSections.length-1){setSection(section+1);setQIndex(0);setWriting(readState().diagnostic.writingText)}else finishAll();
 };

 const saveExit=()=>{persist(s=>{s.diagnostic.status=s.diagnostic.completedSections.length?"in_progress":s.diagnostic.status==="not_started"?"not_started":"in_progress"});navigate("/dashboard")};
 const backSection=()=>{if(qIndex>0){setQIndex(qIndex-1)}else if(section>0){setSection(section-1);setQIndex(0)}else navigate("/dashboard")};

function renderDiagnosticBody(current:any,qIndex:number,state:AppState,setWriting:any,writing:string,answer:any,finishSection:any,audioRef:any,speakingTime:number,setSpeakingTime:any,persist:any){
 if(current.kind==="objective"){const q=current.questions[qIndex];return <div className="question-view"><div className="question-head"><div><div className="eyebrow">{current.name}</div><h2>Question {qIndex+1} of {current.questions.length}</h2></div><Badge tone="gray">{q.skill}</Badge></div><p className="question">{q.q}</p><div className="option-list">{q.options.map((o:string,i:number)=><button key={o} className={`option ${state.diagnostic.responses[q.id]===i?"selected":""}`} onClick={()=>answer(q.id,i)}><span className="option-letter">{String.fromCharCode(65+i)}</span><span>{o}</span></button>)}</div></div>}
 if(current.kind==="reading")return <div className="reading-test-grid"><article className="passage"><div className="eyebrow">Reading passage</div><h2>Urban Green Spaces and City Life</h2>{readingPassage.map((p,i)=><p key={i}>{p}</p>)}</article><div className="question-stack">{current.questions.map((q:any,i:number)=><div className="q-block" key={q.id}><div className="q">{i+1}. {q.q}</div><div className="option-list">{q.options.map((o:string,j:number)=><button className={`option ${state.diagnostic.responses[q.id]===j?"selected":""}`} key={o} onClick={()=>answer(q.id,j)}><span className="option-letter">{String.fromCharCode(65+j)}</span><span>{o}</span></button>)}</div></div>)}</div></div>;
 if(current.kind==="listening")return <div className="listening-test"><div className="audio-box"><audio ref={audioRef} controls preload="metadata" src="/listening-diagnostic.wav" className="audio-native"/><div className="audio-note"><LockKeyhole size={13}/> Transcript hidden during assessment. Play the original prototype passage and answer independently.</div></div><div className="question-stack">{current.questions.map((q:any,i:number)=><div className="q-block" key={q.id}><div className="q">{i+1}. {q.q}</div><div className="option-list">{q.options.map((o:string,j:number)=><button className={`option ${state.diagnostic.responses[q.id]===j?"selected":""}`} key={o} onClick={()=>answer(q.id,j)}><span className="option-letter">{String.fromCharCode(65+j)}</span><span>{o}</span></button>)}</div></div>)}</div></div>;
 if(current.kind==="writing")return <div className="writing-test"><div className="prompt-card"><div className="eyebrow">Writing · Task 2</div><h2>Write your response independently.</h2><blockquote>{writingPrompt}</blockquote><div className="notice top-gap"><strong>Diagnostic mode:</strong> no AI suggestions, rewriting, translation or feedback will be provided while you write.</div><p className="small muted top-gap">Target 180–250 words · 20 minute timer simulated for this prototype.</p></div><div className="editor-wrap"><div className="editor-top"><span>Response</span><span>{wordCount(writing)} words</span></div><textarea className="writing-editor" value={writing} onChange={e=>{setWriting(e.target.value);persist(s=>{s.diagnostic.writingText=e.target.value;s.diagnostic.updatedAt=new Date().toISOString()})}} placeholder="Begin writing here..."/><div className="save-status"><Save size={13}/> Saved locally</div></div></div>;
 if(current.kind==="speaking")return <SpeakingDiagnostic state={state} speakingTime={speakingTime} setSpeakingTime={setSpeakingTime} persist={persist}/>;
 return null;
}

function SpeakingDiagnostic({state,speakingTime,setSpeakingTime,persist}:{state:AppState;speakingTime:number;setSpeakingTime:any;persist:any}){
 const status=state.diagnostic.speakingStatus; const timerRef=useRef<number|undefined>();
 useEffect(()=>{if(status==="PREPARING"||status==="RECORDING"){timerRef.current=window.setInterval(()=>setSpeakingTime((x:number)=>x+1),1000);return()=>{if(timerRef.current)clearInterval(timerRef.current)} }},[status,setSpeakingTime]);
 useEffect(()=>{if(speakingTime>=180&&status==="RECORDING")persist(s=>{s.diagnostic.speakingStatus="SUBMITTED";s.diagnostic.speakingDuration=speakingTime})},[speakingTime,status]);
 const start=()=>persist(s=>{s.diagnostic.speakingStatus="PREPARING";s.diagnostic.speakingDuration=0});
 useEffect(()=>{if(status==="PREPARING"){const t=window.setTimeout(()=>persist(s=>{s.diagnostic.speakingStatus="RECORDING";s.diagnostic.speakingDuration=0}),1000*2);return()=>clearTimeout(t)}},[status,persist]);
 const submit=()=>persist(s=>{s.diagnostic.speakingStatus="SUBMITTED";s.diagnostic.speakingDuration=speakingTime});
 const stateLabel=status==="READY"?"Start when ready":status==="PREPARING"?`Preparing · ${Math.max(0,60-speakingTime)}s`:status==="RECORDING"?`Recording · ${Math.max(0,120-speakingTime)}s remaining`:"Submitted";
 return <div className="speaking-test"><div className={`record-state ${status==="RECORDING"?"recording":""}`}><span className="record-dot"/>{status}</div><div className="speaking-prompt"><div className="eyebrow">Speaking · Part 2</div><h2>{speakingPrompt.title}</h2><ul>{speakingPrompt.bullets.map(x=><li key={x}>{x}</li>)}</ul></div><div className="speaking-timer">{status==="READY"?"1:00":status==="PREPARING"?`${Math.max(0,60-speakingTime)} sec`:status==="RECORDING"?`${Math.max(0,120-speakingTime)} sec`:"Submitted"}</div><p className="muted center">{stateLabel}</p><div className="speaking-actions">{status==="READY"&&<Button onClick={start}><MicIcon/> Start speaking task</Button>}{status==="RECORDING"&&<Button onClick={submit}>Submit response <ArrowRight size={15}/></Button>}{status==="SUBMITTED"&&<SuccessMark label="Recording state saved for this prototype."/ >}</div><div className="notice top-gap"><LockKeyhole size={13}/> Microphone access may be available in a real browser. This prototype stores speaking state and duration locally and uses provisional simulated analysis after submission.</div></div>
}
